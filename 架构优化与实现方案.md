## 1. 总体评价

### ✅ 做得好的地方

1. **技术选型正确**：Kotlin + Compose + Room + MVVM 是当前 Android 开发的主流组合
2. **差异化功能清晰**：悬浮气泡 + 智能识别 + 通知监听形成竞争壁垒
3. **迭代节奏健康**：从 v1.0.0 到 v1.2.1 仅三周，功能迭代快速

### ⚠️ 需要优先解决的问题

1. **金额使用 Double 类型** - 会导致精度问题（最高优先级）
2. **通知监听在小米 HyperOS 上失效** - 用户体验严重受损
3. **智能识别逻辑分散** - 难以维护和测试
4. **宠物系统架构未解耦** - 未来维护困难

## 2. 立即执行：修复金额精度问题（R-001）

### 2.1 问题描述

`Record.amount` 使用 `Double` 类型，会导致浮点精度丢失：

```
0.1 + 0.2 = 0.30000000000000004  // 而非 0.3
38.10 + 12.50 = 50.60000000000001  // 而非 50.60
```

在记账类应用中，金额精度是不可容忍的 Bug。

### 2.2 解决方案

将金额存储单位从「元（Double）」改为「分（Long）」。

```kotlin
// util/AmountExt.kt

/**
 * 金额扩展函数
 * 内部统一使用 Long（分），显示层使用 String（元）
 */

// Long（分）→ 显示字符串
fun Long.toAmountString(): String {
    val yuan = this / 100.0
    return "¥%.2f".format(yuan)
}

// Long（分）→ 纯数字字符串（不含 ¥ 符号）
fun Long.toAmountNumber(): String {
    return "%.2f".format(this / 100.0)
}

// 字符串（元）→ Long（分）
fun String.toAmountCents(): Long {
    val cleaned = this.replace("[^0-9.]".toRegex(), "")
    if (cleaned.isEmpty()) return 0L
    return try {
        (cleaned.toDouble() * 100).toLong()
    } catch (e: NumberFormatException) {
        0L
    }
}

// Double（元）→ Long（分）- 仅用于迁移
fun Double.toCents(): Long = (this * 100).toLong()

// Long（分）→ Double（元）- 仅用于图表等场景
fun Long.toYuanDouble(): Double = this / 100.0
```

#### 2.2.3 Room 数据库迁移

```kotlin
// data/AppDatabase.kt

@Database(
    entities = [RecordEntity::class, CategoryEntity::class],
    version = 3,  // 从 version 2 升级到 version 3
    exportSchema = true
)
abstract class AppDatabase : RoomDatabase() {

    abstract fun recordDao(): RecordDao
    abstract fun categoryDao(): CategoryDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getDatabase(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "jiyixia_database"
                )
                    .addMigrations(MIGRATION_1_2, MIGRATION_2_3)
                    .build()
                INSTANCE = instance
                instance
            }
        }

        /**
         * v1 → v2：添加报销相关字段（如果存在这个版本的话）
         */
        val MIGRATION_1_2 = object : Migration(1, 2) {
            override fun migrate(database: SupportSQLiteDatabase) {
                database.execSQL("""
                    ALTER TABLE records ADD COLUMN isReimbursable INTEGER NOT NULL DEFAULT 0
                """)
                database.execSQL("""
                    ALTER TABLE records ADD COLUMN isReimbursed INTEGER NOT NULL DEFAULT 0
                """)
                database.execSQL("""
                    ALTER TABLE records ADD COLUMN reimbursementTarget TEXT
                """)
            }
        }

        /**
         * v2 → v3：金额类型从 Double(元) 迁移到 Long(分)
         *
         * 策略：新建表 → 数据迁移 → 删旧表 → 重命名
         */
        val MIGRATION_2_3 = object : Migration(2, 3) {
            override fun migrate(database: SupportSQLiteDatabase) {
                // 1. 创建新表
                database.execSQL("""
                    CREATE TABLE records_new (
                        id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                        type INTEGER NOT NULL,
                        amountCents INTEGER NOT NULL,
                        categoryId INTEGER,
                        note TEXT NOT NULL,
                        date INTEGER NOT NULL,
                        createdAt INTEGER NOT NULL,
                        isPendingConfirm INTEGER NOT NULL DEFAULT 0,
                        confidence INTEGER NOT NULL DEFAULT 100,
                        isReimbursable INTEGER NOT NULL DEFAULT 0,
                        isReimbursed INTEGER NOT NULL DEFAULT 0,
                        reimbursementTarget TEXT,
                        FOREIGN KEY (categoryId) REFERENCES categories(id)
                            ON DELETE SET NULL
                    )
                """)

                // 2. 迁移数据：amount * 100 → amountCents
                database.execSQL("""
                    INSERT INTO records_new (
                        id, type, amountCents, categoryId, note,
                        date, createdAt, isPendingConfirm, confidence,
                        isReimbursable, isReimbursed, reimbursementTarget
                    )
                    SELECT
                        id, type,
                        CAST(amount * 100 AS INTEGER),
                        categoryId, note,
                        date, createdAt, isPendingConfirm, confidence,
                        isReimbursable, isReimbursed, reimbursementTarget
                    FROM records
                """)

                // 3. 删除旧表
                database.execSQL("DROP TABLE records")

                // 4. 重命名新表
                database.execSQL("ALTER TABLE records_new RENAME TO records")

                // 5. 重建索引
                database.execSQL("CREATE INDEX index_records_categoryId ON records(categoryId)")
                database.execSQL("CREATE INDEX index_records_date ON records(date)")
                database.execSQL("CREATE INDEX index_records_type ON records(type)")
            }
        }
    }
}
```

## 3. 立即执行：删除无用权限

### 3.1 问题

`AndroidManifest.xml` 中声明了 `INTERNET` 权限，但 APP 当前没有任何网络请求功能。这会：

- 在应用市场审核时被标记为"收集用户数据"的风险项
- 增加用户安装时的心理负担
- 不符合最小权限原则

### 3.2 修改

```xml
<!-- AndroidManifest.xml -->

<!-- ❌ 删除以下权限 -->
<!--
<uses-permission android:name="android.permission.INTERNET" />
-->

<!-- ✅ 当前实际需要的权限 -->
<!-- 悬浮窗权限 -->
<uses-permission android:name="android.permission.SYSTEM_ALERT_WINDOW" />

<!-- 通知监听权限 -->
<uses-permission android:name="android.permission.BIND_NOTIFICATION_LISTENER_SERVICE" />

<!-- 前台服务权限（通知监听保活需要） -->
<uses-permission android:name="android.permission.FOREGROUND_SERVICE" />
<uses-permission android:name="android.permission.FOREGROUND_SERVICE_DATA_SYNC" />

<!-- 通知权限（Android 13+） -->
<uses-permission android:name="android.permission.POST_NOTIFICATIONS" />

<!-- 电池优化白名单请求 -->
<uses-permission android:name="android.permission.REQUEST_IGNORE_BATTERY_OPTIMIZATIONS" />

<!-- 存储权限（仅 Android 9 及以下） -->
<uses-permission
    android:name="android.permission.WRITE_EXTERNAL_STORAGE"
    android:maxSdkVersion="28" />
```

> **注意**：如果后续版本需要云同步功能，届时再加回 `INTERNET` 权限。

#### 4.3.1 金额值对象

```kotlin
// domain/model/Amount.kt

/**
 * 金额值对象
 * 不可变，所有运算返回新实例
 * 内部以「分」为单位的 Long 存储
 */
@JvmInline
value class Amount(val cents: Long) : Comparable<Amount> {

    companion object {
        val ZERO = Amount(0L)

        /** 从元字符串创建： "38.10" → Amount(3810) */
        fun fromYuanString(value: String): Amount {
            val cleaned = value.replace("[^0-9.]".toRegex(), "")
            if (cleaned.isEmpty()) return ZERO
            return try {
                Amount((cleaned.toDouble() * 100).toLong())
            } catch (e: NumberFormatException) {
                ZERO
            }
        }

        /** 从元 Double 创建： 38.10 → Amount(3810) */
        fun fromYuan(value: Double): Amount = Amount((value * 100).toLong())

        /** 从分 Long 创建： 3810L → Amount(3810) */
        fun fromCents(cents: Long): Amount = Amount(cents)
    }

    /** 转为元 Double： Amount(3810) → 38.10 */
    fun toYuanDouble(): Double = cents / 100.0

    /** 转为显示字符串： Amount(3810) → "¥38.10" */
    fun toDisplayString(): String = "¥%.2f".format(cents / 100.0)

    /** 转为纯数字字符串： Amount(3810) → "38.10" */
    fun toNumberString(): String = "%.2f".format(cents / 100.0)

    /** 算术运算 */
    operator fun plus(other: Amount): Amount = Amount(cents + other.cents)
    operator fun minus(other: Amount): Amount = Amount(cents - other.cents)
    operator fun times(multiplier: Int): Amount = Amount(cents * multiplier)

    override fun compareTo(other: Amount): Int = cents.compareTo(other.cents)
}
```

#### 4.3.2 智能识别 UseCase

```kotlin
// domain/usecase/SmartParseUseCase.kt

/**
 * 智能识别 UseCase
 *
 * 统一处理三个入口的输入解析：
 * 1. APP 内文本输入
 * 2. 悬浮气泡文本输入
 * 3. 通知监听自动识别
 *
 * 纯 Kotlin 逻辑，零 Android 依赖，100% 可单元测试
 */
class SmartParseUseCase @Inject constructor(
    private val categoryRepository: CategoryRepository
) {

    data class ParseResult(
        val amount: Amount,
        val categoryId: Long,
        val type: Int,                    // 0=支出，1=收入
        val isReimbursable: Boolean,
        val reimbursementTarget: String?,
        val note: String,
        val confidence: Int               // 0-100
    )

    operator fun invoke(input: String): ParseResult {
        if (input.isBlank()) {
            return emptyResult()
        }

        val amount = extractAmount(input)
        val (categoryId, type) = categorize(input)
        val reimbursement = extractReimbursement(input)
        val note = generateNote(input)
        val confidence = calculateConfidence(input, amount, categoryId)

        return ParseResult(
            amount = amount,
            categoryId = categoryId,
            type = type,
            isReimbursable = reimbursement.isReimbursable,
            reimbursementTarget = reimbursement.target,
            note = note,
            confidence = confidence
        )
    }

    // ========== 金额提取 ==========

    private fun extractAmount(input: String): Amount {
        val patterns = listOf(
            Regex("""(\d+\.?\d*)\s*元"""),
            Regex("""[¥￥]\s*(\d+\.?\d*)"""),
            Regex("""(\d+\.?\d*)""")
        )

        for (pattern in patterns) {
            val match = pattern.find(input)
            if (match != null) {
                return Amount.fromYuanString(match.groupValues[1])
            }
        }

        return Amount.ZERO
    }

    // ========== 分类识别 ==========

    private data class CategoryResult(
        val categoryId: Long,
        val type: Int
    )

    private fun categorize(input: String): CategoryResult {
        // 收入关键词
        val incomeKeywords = mapOf(
            "工资" to "工资", "薪水" to "工资",
            "奖金" to "奖金", "年终奖" to "奖金",
            "理财" to "理财", "利息" to "理财",
            "兼职" to "兼职",
            "红包" to "红包",
            "报销到账" to "报销", "报销" to "报销",
            "租金" to "租金", "房租" to "租金",
            "退款" to "退款",
            "中奖" to "中奖"
        )

        // 支出关键词
        val expenseKeywords = mapOf(
            "午餐" to "餐饮", "晚餐" to "餐饮", "早餐" to "餐饮",
            "外卖" to "餐饮", "餐厅" to "餐饮", "饭" to "餐饮",
            "打车" to "交通", "地铁" to "交通", "公交" to "交通",
            "出租车" to "交通", "加油" to "交通", "停车" to "交通",
            "购物" to "购物", "超市" to "购物", "淘宝" to "购物",
            "房租" to "居住", "水电" to "居住", "物业" to "居住",
            "电影" to "娱乐", "游戏" to "娱乐", "KTV" to "娱乐",
            "医院" to "医疗", "药" to "医疗",
            "话费" to "通讯", "流量" to "通讯"
        )

        // 优先匹配收入
        for ((keyword, categoryName) in incomeKeywords) {
            if (input.contains(keyword)) {
                val category = categoryRepository.findByName(categoryName, type = 1)
                if (category != null) {
                    return CategoryResult(category.id, 1)
                }
            }
        }

        // 匹配支出
        for ((keyword, categoryName) in expenseKeywords) {
            if (input.contains(keyword)) {
                val category = categoryRepository.findByName(categoryName, type = 0)
                if (category != null) {
                    return CategoryResult(category.id, 0)
                }
            }
        }

        // 默认分类
        val defaultCategory = categoryRepository.findByName("其他", type = 0)
        return CategoryResult(defaultCategory?.id ?: 0L, 0)
    }

    // ========== 报销识别 ==========

    private data class ReimbursementResult(
        val isReimbursable: Boolean,
        val target: String?
    )

    private fun extractReimbursement(input: String): ReimbursementResult {
        // "报销到账" 是收入，不标记为可报销
        if (input.contains("报销到账") || input.contains("报销已到账")) {
            return ReimbursementResult(false, null)
        }

        // "XX分公司报销" / "XX公司报销"
        if (input.contains("报销")) {
            val companyPattern = Regex("""([\u4e00-\u9fa5]+(?:公司|分公司|部门))\s*报销""")
            val match = companyPattern.find(input)
            return ReimbursementResult(
                isReimbursable = true,
                target = match?.groupValues?.get(1)
            )
        }

        return ReimbursementResult(false, null)
    }

    // ========== 备注生成 ==========

    private fun generateNote(input: String): String {
        var note = input
            .replace(Regex("""\d+\.?\d*\s*元?"""), "")   // 移除金额
            .replace(Regex("""[¥￥]\s*\d+\.?\d*"""), "")  // 移除 ¥金额
            .replace("报销", "")                           // 移除报销关键词
            .replace(Regex("""[\u4e00-\u9fa5]+(?:公司|分公司|部门)\s*"""), "") // 移除公司名
            .trim()

        if (note.isBlank()) {
            note = ""
        }

        return note
    }

    // ========== 置信度计算 ==========

    private fun calculateConfidence(
        input: String,
        amount: Amount,
        categoryId: Long
    ): Int {
        var score = 0

        // 有金额 +40
        if (amount > Amount.ZERO) score += 40

        // 有分类 +30
        val defaultCategory = categoryRepository.findByName("其他", type = 0)
        if (categoryId != (defaultCategory?.id ?: 0L)) score += 30

        // 输入长度合理 +10
        if (input.length in 2..30) score += 10

        // 包含常见消费关键词 +20
        val commonKeywords = listOf(
            "午餐", "晚餐", "早餐", "外卖",
            "打车", "地铁", "公交",
            "购物", "超市",
            "工资", "奖金"
        )
        if (commonKeywords.any { input.contains(it) }) score += 20

        return score.coerceIn(0, 100)
    }

    private fun emptyResult(): ParseResult {
        val defaultCategory = categoryRepository.findByName("其他", type = 0)
        return ParseResult(
            amount = Amount.ZERO,
```

## 5. 核心问题：小米澎湃3通知监听失效解决方案

### 5.1 问题根因分析

小米15搭载的澎湃3（HyperOS 2 / Android 15 底层）对后台进程管控极其严格，通知监听失效的原因通常是以下组合：

| 序号 | 原因 | 影响程度 |
|------|------|----------|
| 1 | 通知使用权未真正授予（小米有两层权限） | 🔴 高 |
| 2 | 自启动权限未开启，系统杀进程后无法恢复 | 🔴 高 |
| 3 | 电池优化未关闭，系统 Doze 模式冻结 Service | 🟡 中 |
| 4 | 省电模式下前台 Service 被限制 | 🟡 中 |
| 5 | NotificationListenerService 未处理 onListenerDisconnected | 🟡 中 |

```kotlin
// service/PaymentNotificationListener.kt

@AndroidEntryPoint
class PaymentNotificationListener : NotificationListenerService() {

    @Inject lateinit var smartParseUseCase: SmartParseUseCase
    @Inject lateinit var recordRepository: RecordRepository

    private val scope = CoroutineScope(Dispatchers.IO + SupervisorJob())

    override fun onListenerConnected() {
        super.onListenerConnected()
        Log.d(TAG, "NotificationListener connected successfully")

        // 持久化连接状态
        saveConnectionState(true)

        // 启动 Keep-Alive Service
        startKeepAliveService()
    }

    override fun onListenerDisconnected() {
        super.onListenerDisconnected()
        Log.w(TAG, "NotificationListener disconnected")

        // 持久化断开状态
        saveConnectionState(false)

        // HyperOS 断开时自动请求重连
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            requestRebind(
                ComponentName(this, PaymentNotificationListener::class.java)
            )
        }
    }

    override fun onNotificationPosted(sbn: StatusBarNotification?) {
        sbn ?: return

        // 只处理支付类 APP 的通知
        if (sbn.packageName !in PAYMENT_PACKAGES) return

        val notification = sbn.notification ?: return
        val extras = notification.extras ?: return

        // 提取通知文本
        val title = extras.getString(Notification.EXTRA_TITLE).orEmpty()
        val text = extras.getCharSequence(Notification.EXTRA_TEXT)?.toString().orEmpty()
        val subText = extras.getCharSequence(Notification.EXTRA_SUB_TEXT)?.toString().orEmpty()
        val bigText = extras.getCharSequence(Notification.EXTRA_BIG_TEXT)?.toString().orEmpty()

        // 合并所有文本，提高识别率
        val combinedText = listOf(title, text, subText, bigText)
            .filter { it.isNotBlank() }
            .joinToString(" ")

        if (combinedText.isBlank()) return

        handlePaymentNotification(combinedText, sbn.packageName)
    }

    override fun onNotificationRemoved(sbn: StatusBarNotification?) {
        // 不需要处理通知移除
    }

    private fun handlePaymentNotification(
        text: String,
        packageName: String
    ) {
        // 快速过滤：是否包含支付相关关键词
        if (!isPaymentNotification(text)) return

        Log.d(TAG, "Detected payment notification from ${getAppName(packageName)}: $text")

        // 去重检查：同一通知5秒内不重复处理
        val dedupeKey = "${packageName}_${text.hashCode()}"
        if (isRecentlyProcessed(dedupeKey)) {
            Log.d(TAG, "Duplicate notification, skipping")
            return
        }
        markProcessed(dedupeKey)

        // 使用 SmartParseUseCase 统一解析
        val parseResult = smartParseUseCase(text)

        if (parseResult.amount == Amount.ZERO) {
            Log.w(TAG, "Could not extract amount from notification")
            return
        }

        // 插入数据库
        val record = RecordEntity(
            type = parseResult.type,
            amountCents = parseResult.amount.cents,
            categoryId = parseResult.categoryId,
            note = parseResult.note.ifBlank {
                "${getAppName(packageName)}消费"
            },
            date = System.currentTimeMillis(),
            createdAt = System.currentTimeMillis(),
            isPendingConfirm = true,  // 自动记账默认待确认
            confidence = (parseResult.confidence * 0.8).toInt(),  // 自动识别降低置信度
            isReimbursable = parseResult.isReimbursable,
            reimbursementTarget = parseResult.reimbursementTarget
        )

        scope.launch {
            val id = recordRepository.insert(record)

            // 发布领域事件
            DomainEvents.emit(
                DomainEvent.RecordAdded(
                    recordId = id,
                    amountCents = record.amountCents,
                    type = record.type,
                    categoryId = record.categoryId ?: 0L,
                    isAutoDetected = true,
                    isReimbursable = record.isReimbursable
                )
            )

            // 发送待确认通知
            sendConfirmNotification(record)
        }
    }

    /**
     * 判断是否为支付通知
     */
    private fun isPaymentNotification(text: String): Boolean {
        val keywords = listOf(
            // 微信
            "付款成功", "支付成功", "微信支付", "转账-来自",
            // 支付宝
            "付款成功", "收款到账", "余额变动", "消费提醒",
            // 通用
            "扣款成功", "交易成功", "订单支付",
            "你向", "你从", "成功支付"
        )
        return keywords.any { text.contains(it) }
    }

    /**
     * 去重：记录最近5秒内处理过的通知
     */
    private val recentlyProcessed = mutableMapOf<String, Long>()

    private fun isRecentlyProcessed(key: String): Boolean {
        val now = System.currentTimeMillis()
        val lastTime = recentlyProcessed[key]
        return lastTime != null && (now - lastTime) < 5000
    }

    private fun markProcessed(key: String) {
        val now = System.currentTimeMillis()
        recentlyProcessed[key] = now

        // 清理5秒前的记录
        recentlyProcessed.entries.removeIf { now - it.value > 5000 }
    }

    /**
     * 发送待确认通知
     */
    private fun sendConfirmNotification(record: RecordEntity) {
        val channelId = "auto_record_channel"

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                channelId,
                "自动记账",
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply {
                description = "自动识别的消费记录待确认"
            }
            getSystemService(NotificationManager::class.java)
                ?.createNotificationChannel(channel)
        }

        val intent = Intent(this, Class.forName("com.jiyixia.app.MainActivity")).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP
            putExtra("openPendingConfirm", true)
        }

        val pendingIntent = PendingIntent.getActivity(
            this,
            record.id.toInt(),
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(this, channelId)
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle("检测到消费")
            .setContentText("${record.amountCents.toAmountString()} - 待确认")
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()

        getSystemService(NotificationManager::class.java)
            ?.notify(record.id.toInt(), notification)
    }

    /**
     * 持久化连接状态
     */
    private fun saveConnectionState(connected: Boolean) {
        getSharedPreferences("notification_listener", MODE_PRIVATE)
            .edit()
            .putBoolean("is_connected", connected)
            .putLong("last_state_change", System.currentTimeMillis())
            .apply()
    }

    private fun startKeepAliveService() {
        try {
            val intent = Intent(this, NotificationKeepAliveService::class.java)
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                startForegroundService(intent)
            } else {
                startService(intent)
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to start keep-alive service", e)
        }
    }

    private fun getAppName(packageName: String): String {
        return when (packageName) {
            "com.tencent.mm" -> "微信"
            "com.eg.android.AlipayGphone" -> "支付宝"
            "com.unionpay" -> "云闪付"
            else -> packageName.substringAfterLast(".")
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        scope.cancel()
    }

    companion object {
        private const val TAG = "PaymentListener"

        /**
         * 监听的支付 APP 包名
         */
        val PAYMENT_PACKAGES = setOf(
            "com.tencent.mm",              // 微信
            "com.eg.android.AlipayGphone", // 支付宝
            "com.unionpay",                // 云闪付
            "com.samsung.android.spay",    // Samsung Pay
            "com.huawei.wallet"            // 华为钱包
        )
    }
```

```kotlin
// service/NotificationKeepAliveService.kt

/**
 * 通知监听保活 Service
 *
 * 运行在后台的前台 Service，确保 NotificationListenerService
 * 在小米 HyperOS / 华为 EMUI / OPPO ColorOS 等系统上稳定运行
 */
@AndroidEntryPoint
class NotificationKeepAliveService : Service() {

    private var isRunning = false

    override fun onCreate() {
        super.onCreate()
        isRunning = true
        startForegroundNotification()
        Log.d(TAG, "Keep-alive service started")
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (!isRunning) {
            startForegroundNotification()
            isRunning = true
        }

        // 处理来自 MainActivity 的操作指令
        when (intent?.action) {
            ACTION_CHECK_STATUS -> checkAndRestartListener()
            ACTION_STOP -> stopSelf()
        }

        // START_STICKY: 被系统杀死后自动重启
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onTaskRemoved(rootIntent: Intent?) {
        super.onTaskRemoved(rootIntent)
        Log.w(TAG, "Task removed, scheduling service restart")

        // 应用被划掉后，延迟重启 Service
        val restartIntent = Intent(this, NotificationKeepAliveService::class.java)
        val pendingIntent = PendingIntent.getService(
            this, 1, restartIntent,
            PendingIntent.FLAG_ONE_SHOT or PendingIntent.FLAG_IMMUTABLE
        )

        val alarmManager = getSystemService(Context.ALARM_SERVICE) as AlarmManager
        alarmManager.set(
            AlarmManager.ELAPSED_REALTIME_WAKEUP,
            SystemClock.elapsedRealtime() + 3000,
            pendingIntent
        )
    }

    override fun onDestroy() {
        super.onDestroy()
        isRunning = false
        Log.d(TAG, "Keep-alive service destroyed")
    }

    /**
     * 启动前台通知
     */
    private fun startForegroundNotification() {
        createNotificationChannel()

        val notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("记一下")
            .setContentText("自动记账监听运行中")
            .setSmallIcon(R.drawable.ic_notification)
            .setPriority(NotificationCompat.PRIORITY_MIN)
            .setOngoing(true)
            .setSilent(true)
            .setShowWhen(false)
            .build()

        startForeground(NOTIFICATION_ID, notification)
    }

    /**
     * 创建通知渠道
     */
    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "自动记账保活",
                NotificationManager.IMPORTANCE_MIN
            ).apply {
                description = "保持自动记账功能正常运行"
                setShowBadge(false)
                enableLights(false)
                enableVibration(false)
                setSound(null, null)
                lockscreenVisibility = Notification.VISIBILITY_SECRET
            }
            getSystemService(NotificationManager::class.java)
                ?.createNotificationChannel(channel)
        }
    }

    /**
     * 检查并重启通知监听
     */
    private fun checkAndRestartListener() {
        val isConnected = getSharedPreferences("notification_listener", MODE_PRIVATE)
            .getBoolean("is_connected", false)

        if (!isConnected) {
            Log.w(TAG, "Notification listener disconnected, attempting restart")
            // 系统会在 NotificationListenerService.onListenerDisconnected 中处理重连
            // 这里只做状态检查和日志记录
        }
    }

    companion object {
        private const val TAG = "KeepAliveService"
        private const val CHANNEL_ID = "keep_alive_channel"
        private const val NOTIFICATION_ID = 1001

        const val ACTION_CHECK_STATUS = "com.jiyixia.app.CHECK_STATUS"
        const val ACTION_STOP = "com.jiyixia.app.STOP_KEEP_ALIVE"
    }
```

```kotlin
/**
 * 小米 HyperOS 专属权限引导

 * 澎湃系统有以下额外限制需要手动引导用户开启：

object MiuiPermissionHelper {

    private const val TAG = "MiuiPermissionHelper"

    /**
     * 检测是否为小米/Redmi 设备
     */
```

```kotlin
// domain/usecase/PetBehaviorUseCase.kt

/**
 * 宠物行为 UseCase
 * 根据领域事件计算宠物的情绪和反应
 */
class PetBehaviorUseCase @Inject constructor(
    private val recordRepository: RecordRepository
) {

    data class PetState(
        val mood: PetMood,
        val action: PetAction,
        val message: String,
        val growthStage: GrowthStage,
        val totalRecords: Int
    )

    enum class PetMood {
        HAPPY,       // 😊 今天已记账
        NORMAL,      // 😐 昨天记了，今天还没
        SLEEPY,      // 😴 连续2天没记
        SAD          // 😢 连续3天+没记
    }

    enum class PetAction {
        IDLE,           // 正常站立
        JUMP,           // 跳一下
        SPIN,           // 开心转圈
        THUMBS_UP,      // 竖大拇指
        YAWN,           // 打哈欠
        CRY             // 流泪
    }

    enum class GrowthStage(val minRecords: Int, val label: String) {
        BABY(0, "幼年 🐣"),
        TEEN(31, "少年 🐥"),
        ADULT(101, "成年 🐔"),
        EVOLVED(301, "进阶 🦅")
    }

    /**
     * 记账成功时计算宠物反应
     */
    suspend fun onRecordAdded(
        amountCents: Long,
        type: Int,
        isAutoDetected: Boolean
    ): PetState {
        val totalRecords = recordRepository.getTotalCount()
        val stage = calculateGrowthStage(totalRecords)

        val (action, message) = when {
            type == 1 && amountCents > 1000000 -> {
                // 大额收入（>10000元）
                PetAction.SPIN to "发财啦！🎉"
            }
            type == 1 -> {
                PetAction.SPIN to "有钱啦！💰"
            }
            isAutoDetected -> {
                PetAction.JUMP to "帮你记下了！📝"
            }
            else -> {
                PetAction.JUMP to "谢谢主人！💬"
            }
        }

        return PetState(
            mood = PetMood.HAPPY,
            action = action,
            message = message,
            growthStage = stage,
            totalRecords = totalRecords
        )
    }

    /**
     * 计算当前宠物情绪（基于记账连续性）
     */
    suspend fun calculateCurrentMood(): PetState {
        val streak = calculateStreak()
        val totalRecords = recordRepository.getTotalCount()
        val stage = calculateGrowthStage(totalRecords)

        val (mood, action, message) = when {
            streak == 0 -> {
                // 今天已记账
                PetMood.HAPPY to PetAction.IDLE to "今天很棒！😊"
            }
            streak == 1 -> {
                PetMood.NORMAL to PetAction.IDLE to "今天还没记账哦"
            }
            streak == 2 -> {
                PetMood.SLEEPY to PetAction.YAWN to "两天没记账了...😴"
            }
            else -> {
                PetMood.SAD to PetAction.CRY to "好久没记账了😢"
            }
        }

        return PetState(
            mood = mood,
            action = action,
            message = message,
            growthStage = stage,
            totalRecords = totalRecords
        )
    }

    private fun calculateGrowthStage(totalRecords: Int): GrowthStage {
        return GrowthStage.entries
            .lastOrNull { totalRecords >= it.minRecords }
            ?: GrowthStage.BABY
    }

    /**
     * 计算连续未记账天数
     * 返回0表示今天已记账
     */
    private suspend fun calculateStreak(): Int {
        // 简化实现：检查最近3天的记录数
        val today = startOfDay(System.currentTimeMillis())
        var streak = 0

        for (daysAgo in 0..3) {
            val dayStart = today - (daysAgo * 86400000L)
            val dayEnd = dayStart + 86400000L
            val records = recordRepository.getCountByDateRange(dayStart, dayEnd)

            if (records > 0) {
                break
            }
            streak++
        }

        return streak
    }

    private fun startOfDay(timestamp: Long): Long {
        val calendar = java.util.Calendar.getInstance()
        calendar.timeInMillis = timestamp
        calendar.set(java.util.Calendar.HOUR_OF_DAY, 0)
        calendar.set(java.util.Calendar.MINUTE, 0)
        calendar.set(java.util.Calendar.SECOND, 0)
        calendar.set(java.util.Calendar.MILLISECOND, 0)
        return calendar.timeInMillis
    }
}
```

### 7.1 优先级矩阵

| 优先级 | 任务 | 预计时间 | 影响范围 | 说明 |
|--------|------|----------|----------|------|
| 🔴 P0 | 修复金额 Double → Long | 1-2天 | 全局 | 数据正确性，越早越好 |
| 🔴 P0 | 小米通知监听修复（5层方案） | 2-3天 | 核心功能 | 当前完全失效 |

```kotlin
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): AppDatabase {

            "jiyixia_database"
```
